{公式名：动量反弹选股}
{用途：超跌/弱市反弹。MACD零轴下，绿柱缩短/拐头/金叉任一出现即可；ROE>0；成交额自定义}
{2026-08-16 按主人选股方法论编写；2026-08-17 补「均线拐头5.10.20」加码档（对齐阿狸咪初衷截图）}

{━━ 可调参数（按盘面改） ━━}
成交额门槛:=1000000000;   {成交额(元)，默认10亿。行情差降到3-5亿，行情好升到15-20亿}
超跌幅度:=5;              {跌破5日线幅度%，超跌反弹用；普通反弹可设0}
波动阈值:=12;             {10日振幅%，要求波动大；弱势震荡市可降到8}
底背离窗口:=20;           {底背离判断周期}

{━━ 指标 ━━}
DIF:=EMA(CLOSE,12)-EMA(CLOSE,26);
DEA:=EMA(DIF,9);
MACD柱:=(DIF-DEA)*2;
MA5:=MA(CLOSE,5);
MA10:=MA(CLOSE,10);
MA20:=MA(CLOSE,20);
零轴下:=DIF<0 AND DEA<0;
绿柱缩短:=MACD柱<0 AND MACD柱>REF(MACD柱,1);   {绿柱缩短=空头动能衰竭}
拐头:=DIF>REF(DIF,1) AND REF(DIF,1)<REF(DIF,2);  {DIF拐头向上}
金叉:=CROSS(DIF,DEA);                             {DIF上穿DEA}
ROE正:=FINANCE(33)>0;                            {净资产收益率为正；如你的通达信版本FINANCE(33)不是ROE，改FINANCE(35)}
量不再萎缩:=VOL>=REF(VOL,1) OR VOL>=MA(VOL,5);   {成交量不再萎缩}

{━━ 超跌/波动条件 ━━}
超跌:=CLOSE<MA5*(1-超跌幅度/100);
波动大:=(HHV(HIGH,10)-LLV(LOW,10))/LLV(LOW,10)*100>=波动阈值;

{━━ 底背离（加分项，形成则胜率加倍） ━━}
价格新低:=LOW<=LLV(LOW,底背离窗口);
MACD未新低:=MACD柱>REF(LLV(MACD柱,底背离窗口),1);
底背离:=价格新低 AND MACD未新低;

{━━ 均线拐头（反弹变强加码，阿狸咪"反弹强一点就是均线拐头，看快慢就是5.10.20"） ━━}
均线拐头快:=MA5>REF(MA5,1) AND MA10>REF(MA10,1);                      {5/10日线同步拐头=反弹变强(快慢看短)}
均线拐头全:=MA5>REF(MA5,1) AND MA10>REF(MA10,1) AND MA20>REF(MA20,1); {5/10/20 全拐头=更强确认}

{━━ 组合输出 ━━}
反弹信号:=零轴下 AND (绿柱缩短 OR 拐头 OR 金叉) AND ROE正 AND 量不再萎缩 AND AMOUNT>=成交额门槛;

{★ 输出1：宽松反弹（行情好放宽，只要求跌破5日线，不给波动阈值）}
宽松反弹:反弹信号 AND 超跌;

{★ 输出2：标准反弹（含超跌+波动要求，弱市用）}
标准反弹:反弹信号 AND 超跌 AND 波动大;

{★ 输出3：均线拐头反弹（反弹变强确认，阿狸咪"均线拐头5.10.20"加码档）}
均线拐头反弹:反弹信号 AND 均线拐头快;

{★ 输出4：底背离反弹（胜率加倍口径，最严）}
底背离反弹:反弹信号 AND 底背离;

{用法：四档由松到严：宽松反弹→标准反弹→均线拐头反弹(反弹变强)→底背离反弹(胜率加倍)。
 行情好看宽松/标准；想确认反弹强度看均线拐头；求最严胜率看底背离。成交额门槛按盘面调}
